⚡ Automatic Real-Time Sync with Google Forms

Whenever a participant submits your Google Form, this script will automatically push their submission directly into this EMBERS Management database without manual copy-pasting.

1

Open your Google Form or Linked Google Sheet

Go to your Google Form: forms.gle/FydTB5VYcbqmsBbj6 ↗, open the Responses tab, and click View in Sheets.

2

Open Apps Script

In the Google Sheet menu, click Extensions > Apps Script.

3

Paste this Script & Save

Delete existing code in the editor, paste the script below, and click Save (💾 icon).

function onFormSubmit(e) {
  // EMBERS '26 Real-Time Google Form Sync Webhook
  var endpoint = "https://embedthon-judge-portal.vercel.app/api/embers";
  
  var values = e ? e.namedValues : {};
  var record = {
    timestamp: values["Timestamp"] ? values["Timestamp"][0] : new Date().toISOString(),
    track: values["Event Track"] ? values["Event Track"][0] : "Track 1: Embeddathon",
    title: values["Project / Paper Title"] ? values["Project / Paper Title"][0] : "Untitled",
    college: values["College Name"] ? values["College Name"][0] : "PACE",
    dept: values["Department & Year/Semester"] ? values["Department & Year/Semester"][0] : "CSE",
    leaderName: values["Team Leader Name"] ? values["Team Leader Name"][0] : "Leader",
    email: values["Email Address"] ? values["Email Address"][0] : "",
    phone: values["WhatsApp / Phone Number"] ? values["WhatsApp / Phone Number"][0] : "",
    usn: values["College USN / Roll No"] ? values["College USN / Roll No"][0] : "",
    membersCount: values["Total Number of Team Members (including leader)"] ? parseInt(values["Total Number of Team Members (including leader)"][0]) : 1,
    member2: values["Member 2 Name & USN"] ? values["Member 2 Name & USN"][0] : "",
    member3: values["Member 3 Name & USN"] ? values["Member 3 Name & USN"][0] : "",
    member4: values["Member 4 Name & USN"] ? values["Member 4 Name & USN"][0] : "",
    abstract: values["Brief Abstract / Description (50-100 words)"] ? values["Brief Abstract / Description (50-100 words)"][0] : "",
    domain: values["Category / Domain"] ? values["Category / Domain"][0] : "Embedded Systems",
    hardwareReq: values["Hardware / Power Requirement (if any)"] ? values["Hardware / Power Requirement (if any)"][0] : "None"
  };

  var payload = {
    action: "sync_responses",
    records: [record]
  };

  var options = {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  };

  UrlFetchApp.fetch(endpoint, options);
}
4

Add Trigger

Click the clock icon (Triggers) on the left sidebar > + Add Trigger > Select function: onFormSubmit, Event type: On form submit > Click Save. You're done!
